## 4. 実行コマンド例
```bash
# 120 営業日候補データ生成（複数閾値＆追記）
PYTHONPATH=. python analysis/build_multi_model_candidate_dataset.py \
  --lookback-days 120 --max-candidates 150 \
  --temp-dir tmp/multi_candidate_builder --append-output \
  --down-thresholds=-0.01,-0.015,-0.02

# Precision レポート（フォールバック無効）
PYTHONPATH=. python analysis/multi_model_precision_report.py \
  --input production_data/multi_model_candidates.parquet \
  --fallback-max 0

# Precision レポート（フォールバック比率 0.4）
PYTHONPATH=. python analysis/multi_model_precision_report.py \
  --input production_data/multi_model_candidates.parquet \
  --allow-fallback --fallback-max 1 \
  --fallback-min-passed 2 --fallback-min-passed-ratio 0.4

# 閾値サーチ（fallback 比率付き）
PYTHONPATH=. python analysis/multi_model_threshold_optimizer.py \
  --input production_data/multi_model_candidates.parquet \
  --threshold-up-grid 0.16,0.17,0.18,0.19,0.20 \
  --threshold-down-grid 0.48,0.50,0.52 \
  --risk-grid 0.35,0.40,0.45 \
  --weight-up 1.0 --weight-down 0.5 --weight-risk 0.4 \
  --top-n 5 --metric precision \
  --metric-weights precision:0.6,avg_net_return:0.3,coverage_rate:0.1 \
  --allow-fallback --fallback-max 1 \
  --fallback-min-passed 2 --fallback-min-passed-ratio 0.4 \
  --export-csv analysis/multi_model_threshold_grid_120d.csv

# 閾値サーチ（フォールバック無効）
PYTHONPATH=. python analysis/multi_model_threshold_optimizer.py \
  --input production_data/multi_model_candidates.parquet \
  --threshold-up-grid 0.18,0.19,0.20,0.21,0.22 \
  --threshold-down-grid 0.48,0.50,0.52 \
  --risk-grid 0.35,0.40,0.45 \
  --top-n 5 --metric precision --fallback-max 0 \
  --export-csv analysis/multi_model_threshold_grid_120d_no_fallback.csv
```
